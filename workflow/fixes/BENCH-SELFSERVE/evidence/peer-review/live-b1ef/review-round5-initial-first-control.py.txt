retained = {f['sha256'] for f in files}
documents = []

    assert isinstance(event.get('message_id'), int)
    for doc in event.get('files', []):
        method = 'retained_raw_file'
        if doc['sha256'] not in retained:
            payload = reconstructed[doc['name']]
            assert sha(payload) == doc['sha256'] and len(payload) == doc['bytes']
            method = 'byte_exact_reconstruction_from_retained_prequeue_validation_or_batch'
        documents.append({**doc, 'verified_by': method})
