package main

import (
	"bytes"
	"encoding/json"
	"os"
	"text/template"
)

func main() {
	raw, err := os.ReadFile(os.Args[1])
	if err != nil {
		panic(err)
	}
	t, err := template.New("exact-proposal").Option("missingkey=error").Parse(string(raw))
	if err != nil {
		panic(err)
	}
	out := []map[string]any{}
	for _, env := range []string{"staging", "production", "dev", "stage"} {
		for _, service := range []string{"brain", "manager", "console", "actuator", "sensor", "attacks"} {
			chart := "deploy/charts/phorvex-" + service
			if service == "manager" {
				chart = "phorvex/charts/phorvex"
			}
			params := map[string]string{"environment": env, "service": service, "chart": chart, "release": "staging"}
			var b bytes.Buffer
			if err := t.Execute(&b, params); err != nil {
				panic(err)
			}
			out = append(out, map[string]any{"params": params, "patch": b.String()})
		}
	}
	var b bytes.Buffer
	if t.Execute(&b, map[string]string{"environment": "staging"}) == nil {
		panic("missingkey did not refuse")
	}
	if err := json.NewEncoder(os.Stdout).Encode(out); err != nil {
		panic(err)
	}
}
